\RequirePackage{eso-pic}
\newlength{\cvhero}\setlength{\cvhero}{6.05cm}

\AddToShipoutPictureBG*{\AtPageUpperLeft{%
    \begin{tikzpicture}[overlay]
        \fill[cvname] (0,0) rectangle (\paperwidth,-\cvhero);
        \begin{scope}
            \clip (0,0) rectangle (\paperwidth,-\cvhero);
            % a soft glow: concentric circles from navy to blue (no transparency needed)
            \foreach \i in {0,...,24}{
                \pgfmathsetmacro{\r}{7.6-\i*0.29}
                \pgfmathsetmacro{\m}{\i*2.6}
                \fill[cvaccent!\m!cvname] (\paperwidth-0.6cm,0.4cm) circle (\r cm);}
            \foreach \x in {0,...,9}{\foreach \y in {0,...,9}{
                \fill[cvpale!50!cvname] (\paperwidth-1.85cm-\x*0.42cm,-1.55cm-\y*0.42cm) circle (0.6pt);}}
        \end{scope}
    \end{tikzpicture}}}
